1. 你可以在同一个进程中运行多个 Gin 服务器——每个服务器在不同的端口上——通过使用 `golang.org/x/sync/errgroup` 包中的 `errgroup.Group`。当你需要暴露不同的 API 时（例如，在端口 8080 上的公共 API 和端口 8081 上的管理 API），这非常有用，无需部署单独的二进制文件。每个服务器拥有自己的路由器、中间件栈和 `http.Server` 配置。
2. 示例
``` go
var g errgroup.Group //errgroup.Group 来自 golang.org/x/sync/errgroup，用于并发执行多个 goroutine，并在任一 goroutine 出错时收集错误、取消其余 goroutine。

func router1() http.Handler {  
    r1 := gin.Default()  
    r1.GET("/", func(c *gin.Context) {  
       c.JSON(200, gin.H{  
          "msg": "ok1",  
       })  
    })  
    return r1  
}  

func router2() http.Handler {  
    r2 := gin.Default()  
    r2.GET("/", func(c *gin.Context) {  
       c.JSON(200, gin.H{  
          "msg": "ok2",  
       })  
    })  
    return r2  
}  

func main() {  
    s1 := &http.Server{  
       Addr:         ":8081",  
       Handler:      router1(),  
       ReadTimeout:  3 * time.Second,  
       WriteTimeout: 3 * time.Second,  
    }  

    s2 := &http.Server{  
       Addr:         ":8082",  
       Handler:      router2(),  
       ReadTimeout:  3 * time.Second,  
       WriteTimeout: 3 * time.Second,  
    }  

    g.Go(func() error {  
       return s1.ListenAndServe()  
    })  

    g.Go(func() error {  
       return s2.ListenAndServe()  
    })  

    err := g.Wait()  
    if err != nil {  
       log.Fatal(err.Error())  
    }  
}
```
