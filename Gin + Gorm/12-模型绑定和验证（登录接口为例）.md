1. 绑定标签（Binding Tags）是写在结构体字段后面的**反引号标签**，用来告诉 Gin：「这个字段应该从哪里取、要满足什么规则」，支持绑定 JSON、XML、YAML 和标准表单值（foo=bar&boo=baz）。
2. 两种方式
	1. `MustBind`
		1. 方法：`Bind`、`BindJSON`、`BindXML`、`BindQuery`、`BindYAML`
	2. `ShouldBind`(99%使用)
		1. 方法：`ShouldBind`、`ShouldBindJSON`、`ShouldBindXML`、`ShouldBindQuery`、`ShouldBindYAML`
	3. 区别：`ShouldBind` = 绑定失败返回错误，你决定怎么处理；`MustBind` = 绑定失败直接返回 400，不给你处理机会。**永远用 `ShouldBind`**。
3. 可以指定特定字段为必填。如果一个字段标记了 `binding:"required"` 并且在绑定时值为空，将返回错误。
4. 示例
``` GO
type Login struct {  
    Name     string `json:"name" binding:"required"`  //这个就是绑定标签
    Password string `json:"password" binding:"required"`  //binding:"required" 表示一定要有值，如果传过来的值为空则直接报错
    
    /* 
	    binding:"-" 和 binding:"required"
	    binding:"-" ：忽略该字段，不参与绑定也不参与验证
	    binding:"required" ：必填，该字段必须提供且不能为零值
    */
}  
  
func main() {  
    r := gin.Default()  
    r.StaticFile("/", "index.html")  
    g1 := r.Group("/user")  
    g1.POST("/login", func(c *gin.Context) {  
       var login Login  
       //给 结构体 login 绑定前端发送的表单数据，要用指针。。
       err := c.ShouldBindJSON(&login)  
       if err != nil {  
          //c.String(400, "请求有误")  
          c.JSON(400, gin.H{  
             "message": "请求有误",  
          })  
          return  
       }  
       name := login.Name  
       password := login.Password  
       fmt.Println(name)  
       fmt.Println(password)  
       if name != "zzy" || password != "123456" {  
          //c.String(http.StatusUnauthorized,"认证失败")  
          c.JSON(http.StatusUnauthorized, gin.H{  
             "message": "认证失败",  
          })  
          return  
       }  
       c.JSON(200, gin.H{  
          "message": "认证成功",  
       })  
    })  
    r.Run()  
}
```
